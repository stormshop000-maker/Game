let chatSocket=null; function initChat(){ chatSocket=io('https://game-chat-bif5.onrender.com') let nickname=localStorage.getItem('tlh_nickname'); if(!nickname){ nickname=prompt('ПОЗЫВНОЙ (ник в эфире):','Сталкер'+Math.floor(Math.random()*999)); if(!nickname||!nickname.trim())nickname='Сталкер'+Math.floor(Math.random()*999); localStorage.setItem('tlh_nickname',nickname); } chatSocket.on('connect',()=>chatSocket.emit('join',nickname)); chatSocket.on('welcome',addChatSystem); chatSocket.on('system',(m)=>{addChatSystem(m);flashChatBtn()}); chatSocket.on('online',(n)=>{document.getElementById('online-count').textContent=n+' в Зоне'}); chatSocket.on('chat message',(d)=>{addChatMsg(d.nick,d.text);flashChatBtn()}); chatSocket.on('disconnect',()=>addChatSystem('Связь потеряна...')); const toggle=document.getElementById('chat-toggle'); const box=document.getElementById('chat-container'); const input=document.getElementById('chat-input'); toggle.addEventListener('click',()=>{box.classList.toggle('open');toggle.classList.remove('alert');if(box.classList.contains('open'))setTimeout(()=>input.focus(),100)}); const send=()=>{const t=input.value.trim();if(!t||!chatSocket)return;chatSocket.emit('chat message',t);input.value=''}; document.getElementById('chat-send').addEventListener('click',send); input.addEventListener('keypress',(e)=>{if(e.key==='Enter')send()}); } function addChatMsg(nick,text){ const box=document.getElementById('chat-messages'); const div=document.createElement('div'); div.className='msg'; div.innerHTML=''+escChat(nick)+': '+escChat(text); box.appendChild(div);box.scrollTop=box.scrollHeight; } function addChatSystem(text){ const box=document.getElementById('chat-messages'); const div=document.createElement('div'); div.className='msg system';div.textContent=text; box.appendChild(div);box.scrollTop=box.scrollHeight; } function escChat(s){return String(s).replace(/[&<>"']/g,c=>({'&':'&','<':'<','>':'>','"':'"',"'":'''}[c]))} function flashChatBtn(){ const btn=document.getElementById('chat-toggle'); if(document.getElementById('chat-container').classList.contains('open'))return; btn.classList.add('alert');setTimeout(()=>btn.classList.remove('alert'),2000); } window.addEventListener('load',()=>setTimeout(initChat,500));
